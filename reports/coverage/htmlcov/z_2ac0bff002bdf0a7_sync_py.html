1"""Daily sync job: calls every fetch_* in this module and upserts the day-keyed results. 

2Each fetcher is run independently — one failure doesn't abort the others. Errors are logged 

3and recorded on the ServiceLog row, but the run is still marked is_success=False so admins 

4can spot partial failures."""

23 "anthropic": [fetch_anthropic_daily_usage],

30class ServiceMonitor: 

31 """Runs all fetchers in sequence and upserts each into its own daily-history table.""" 

32 

33 def __init__(self) -> None: 

34 """Initialise the ServiceMonitor class""" 

35 

36 self.logger = logger.AppLogger.create_service_logger(SERVICE_NAME, "INFO") 

37 

38 def run(self, db: Session | None = None) -> None: 

39 """Fetch the different external services data 

40 :param db: Database session 

41 :return Service log entry""" 

42 

43 db = next(get_db()) if db is None else db 

44 start = dt.datetime.now(dt.timezone.utc) 

45 service_log = ExternalServiceMonitoringServiceLog(run_datetime=start) 

46 db.add(service_log) 

47 db.commit() 

48 db.refresh(service_log) 

49 

50 errors: list[str] = [] 

51 try: 

52 for name in EXTERNAL_SERVICES: 

53 try: 

54 self.logger.info(f"Fetching {name}") 

55 for fetch_fn in EXTERNAL_SERVICES[name]: 

56 fetch_fn(db) 

57 except Exception as exc: 

58 db.rollback() 

59 tb = traceback.format_exc() 

60 self.logger.exception(f"{name} failed: {exc}") 

61 errors.append(f"{name}: {exc}\n{tb}") 

62 finally: 

63 service_log.run_duration = (dt.datetime.now(dt.timezone.utc) - start).total_seconds() 

64 if errors: 

65 service_log.is_success = False 

66 service_log.error_message = "\n---\n".join(errors) 

67 else: 

68 service_log.is_success = True 

69 db.commit() 

70 db.refresh(service_log) 

71 return service_log
