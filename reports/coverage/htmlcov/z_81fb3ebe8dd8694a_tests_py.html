18@payment_test_router.delete("/delete-all-customers") 

19async def delete_all_stripe_customers() -> dict: # pragma: no cover 

20 """Delete ALL Stripe customers, test clocks, and cancel all subscriptions. 

21 WARNING: This permanently deletes ALL customers and test clocks from Stripe 

22 and immediately cancels all active subscriptions across your entire account. 

23 This action cannot be undone. 

24 :return: dict with deletion summary""" 

25 

26 deleted_customers = 0 

27 deleted_clocks = 0 

28 failed_deletions = [] 

29 

30 try: 

31 # Delete all customers with pagination 

32 has_more = True 

33 starting_after = None 

34 

35 while has_more: 

36 params = {"limit": 100} 

37 if starting_after: 

38 params["starting_after"] = starting_after # noqa 

39 

40 customers = await stripe.Customer.list_async(**params) 

41 

42 for customer in customers.data: 

43 try: 

44 deleted_customer = await stripe.Customer.delete_async(customer.id) 

45 if getattr(deleted_customer, "deleted", False): 

46 deleted_customers += 1 

47 logger.info(f"Deleted Stripe customer {customer.id}") 

48 else: 

49 failed_deletions.append(customer.id) 

50 except stripe.error.StripeError as e: 

51 logger.error(f"Failed to delete customer {customer.id}: {str(e)}") 

52 failed_deletions.append(customer.id) 

53 

54 has_more = customers.has_more 

55 if has_more and customers.data: 

56 starting_after = customers.data[-1].id 

57 

58 # Delete all test clocks with pagination 

59 has_more = True 

60 starting_after = None 

61 

62 while has_more: 

63 params = {"limit": 100} 

64 if starting_after: 

65 params["starting_after"] = starting_after # noqa 

66 

67 test_clocks = await stripe.test_helpers.TestClock.list_async(**params) 

68 

69 for clock in test_clocks.data: 

70 try: 

71 await stripe.test_helpers.TestClock.delete_async(clock.id) 

72 deleted_clocks += 1 

73 logger.info(f"Deleted test clock {clock.id}") 

74 except stripe.error.StripeError as e: 

75 logger.error(f"Failed to delete test clock {clock.id}: {str(e)}") 

76 failed_deletions.append(clock.id) 

77 

78 has_more = test_clocks.has_more 

79 if has_more and test_clocks.data: 

80 starting_after = test_clocks.data[-1].id 

81 

82 return {

84 "message": f"Deleted {deleted_customers} customers and {deleted_clocks} test clocks",

91 except stripe.error.StripeError as e:

95 detail="Payment service error. Please try again.",

112@payment_test_router.post("/advance-test-clock") 

113async def advance_test_clock( 

114 request: AdvanceClockRequest, 

115 current_user: User = Depends(get_current_user), 

116) -> dict: # pragma: no cover 

117 """Advance a Stripe test clock by X days for testing subscriptions. 

118 This allows you to simulate time passing to test trial expirations, 

119 billing cycles, and subscription renewals without waiting. 

120 :param request: Request with days to advance and optional test_clock_id 

121 :param current_user: Authenticated user from JWT token 

122 :return: dict with updated test clock information 

123 :raises HTTPException: If test clock doesn't exist or Stripe error occurs""" 

124 

125 try: 

126 # Get test clock ID (from request or retrieve user's test clock) 

127 test_clock_id = request.test_clock_id 

128 

129 if not test_clock_id: 

130 # If no clock ID provided, find the user's customer test clock 

131 if not current_user.stripe_details.customer_id: 

132 raise HTTPException( 

133 status_code=status.HTTP_404_NOT_FOUND, 

134 detail="No Stripe customer found. Create a subscription first.", 

135 ) 

136 

137 # Retrieve customer to get their test clock 

138 customer = await stripe.Customer.retrieve_async(current_user.stripe_details.customer_id) 

139 test_clock_id = getattr(customer, "test_clock", None) 

140 

141 if not test_clock_id: 

142 raise HTTPException( 

143 status_code=status.HTTP_404_NOT_FOUND, 

144 detail="Customer is not associated with a test clock", 

145 ) 

146 

147 # Retrieve current test clock to get frozen_time 

148 test_clock = await stripe.test_helpers.TestClock.retrieve_async(test_clock_id) 

149 

150 if test_clock.status != "ready": 

151 raise HTTPException( 

152 status_code=status.HTTP_409_CONFLICT, 

153 detail=f"Test clock is currently {test_clock.status}. Wait until ready.", 

154 ) 

155 

156 # Calculate new frozen time (current + days) 

157 current_frozen_time = test_clock.frozen_time 

158 seconds_to_advance = request.days * 24 * 60 * 60 

159 new_frozen_time = current_frozen_time + seconds_to_advance 

160 

161 # Advance the test clock 

162 await stripe.test_helpers.TestClock.advance_async(test_clock_id, frozen_time=new_frozen_time) 

163 

164 logger.info(f"Advanced test clock {test_clock_id} by {request.days} days for user {current_user.id}") 

165 

166 # Wait for the test clock to be ready again (poll up to 30 seconds) 

167 max_wait_time = 30 

168 poll_interval = 0.5 

169 elapsed_time = 0 

170 

171 while elapsed_time < max_wait_time: 

172 await asyncio.sleep(poll_interval) 

173 elapsed_time += poll_interval 

174 

175 updated_clock = await stripe.test_helpers.TestClock.retrieve_async(test_clock_id) 

176 if updated_clock.status == "ready": 

177 break 

178 else: 

179 # Timeout waiting for clock to be ready 

180 raise HTTPException( 

181 status_code=status.HTTP_408_REQUEST_TIMEOUT, 

182 detail=f"Test clock did not become ready within {max_wait_time} seconds", 

183 ) 

184 

185 # Retrieve final status 

186 updated_clock = await stripe.test_helpers.TestClock.retrieve_async(test_clock_id) 

187 

188 return { 

189 "success": True, 

190 "test_clock_id": test_clock_id, 

191 "days_advanced": request.days, 

192 "previous_time": dt.datetime.fromtimestamp(current_frozen_time).isoformat(), 

193 "new_time": dt.datetime.fromtimestamp(new_frozen_time).isoformat(), 

194 "status": updated_clock.status, 

195 "message": f"Advanced test clock by {request.days} days", 

196 } 

197 

198 except stripe.error.InvalidRequestError as e: 

199 logger.error(f"Invalid test clock request: {str(e)}") 

200 raise HTTPException( 

201 status_code=status.HTTP_404_NOT_FOUND, 

202 detail="Test clock not found or invalid advancement", 

203 ) 

204 except stripe.error.StripeError as e: 

205 logger.error(f"Stripe error advancing test clock: {str(e)}", exc_info=True) 

206 raise HTTPException( 

207 status_code=status.HTTP_503_SERVICE_UNAVAILABLE, 

208 detail="Payment service temporarily unavailable. Please try again.", 

209 ) 

210 except HTTPException: 

211 raise 

212 except Exception as e: 

213 logger.error(f"Unexpected error advancing test clock: {str(e)}", exc_info=True) 

214 raise HTTPException( 

215 status_code=status.HTTP_500_INTERNAL_SERVER_ERROR, 

216 detail="An error occurred. Please try again.",
