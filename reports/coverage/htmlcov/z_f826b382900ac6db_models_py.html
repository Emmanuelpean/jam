33def get_setting_value(db, name: str, default: Any): 

34 """Retrieve a setting value from the database by its name. 

35 :param db: Database session. 

36 :param name: The name of the setting to retrieve. 

37 :param default: The default value to return if the setting is not found.""" 

38 

39 entry = db.query(Setting).filter(Setting.name == name).filter(Setting.is_active).first()

47 """Represents core user identification and authentication.

49 Attributes:

52 - `email` (str, unique): User's email address. 

53 - `is_active` (bool): Indicates whether the user account is active.

209 

210 @hybrid_property 

211 def is_valid(self) -> bool: 

212 """Check if the token is valid""" 

213 

214 # Define expiration times based on the token type 

215 expiration_minutes = { 

216 "verification": settings.verification_token_expiration_minutes, 

217 "password_reset": settings.password_reset_token_expiration_minutes, 

218 "email_change": settings.email_change_token_expiration_minutes, 

219 } 

220 

221 # noinspection PyTypeChecker 

222 minutes = expiration_minutes.get(self.token_type, settings.verification_token_expiration_minutes) 

223 expiration_time = self.created_at + dt.timedelta(minutes=minutes) 

224 return dt.datetime.now(dt.timezone.utc) < expiration_time 

225 

226 @hybrid_property 

227 def remaining_seconds(self) -> int: 

228 """Calculate how many seconds remain until the next email can be sent. 

229 :return: seconds remaining until next email can be sent""" 

230 

231 time_since_last_email = int((dt.datetime.now(dt.timezone.utc) - self.created_at).total_seconds()) 

232 return math.ceil(settings.verification_email_min_interval_seconds - time_since_last_email)
